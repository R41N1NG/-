```javascript
function parseXianmengFromText(t) {
  const CNM = { 一: 1, 二: 2, 三: 3, 四: 4, 五: 5, 六: 6, 七: 7, 八: 8, 九: 9, 十: 10, 十一: 11, 十二: 12 };
  /* ⚠️ 「日」必须**可省**：玩家写「六月初七」「6月7日」都不该丢日子（首版要求带「日」字 ⇒ 初七被吃掉） */
  const m = /仙盟历\s*(\d{3,4})\s*年\s*[·\.、]?\s*([一二三四五六七八九十]{1,3}|\d{1,2})\s*月\s*(?:[·\.、]?\s*(初[一二三四五六七八九十]|[一二三四五六七八九十]{1,3}|\d{1,2})\s*日?)?/.exec(String(t || ''));
  if (!m) return null;
  const y = Number(m[1]);
  const mo = /^\d+$/.test(m[2]) ? Number(m[2]) : (CNM[m[2]] || 0);
  if (!y || mo < 1 || mo > 12) return null;
  let d = 1;
  const ds = m[3] || '';
  if (ds) {
    if (ds.startsWith('初')) d = CNM[ds.slice(1)] || 1;
    else if (/^\d+$/.test(ds)) d = Number(ds);
    else d = CNM[ds] || 1;
  }
  return { ym: y + mo / 100, day: Math.max(1, Math.min(30, d)) };
}
```
