/**
 * @module okh_utils
 * @updated 2026-11-09
 * @status implemented
 * @reason 通用校验、深拷贝、时段和模块注册
 */
(function (root) {
  'use strict';

  const registry =
    root.OKHBackgroundModules ||= Object.create(null);

  if (registry.utils) {
    throw Error('重复加载 okh_utils');
  }

  const own = (object, name) =>
    Object.hasOwn(object, name);

  const check = (ok, message) => {
    if (!ok) throw Error(message);
  };

  const minute = value =>
    Number.isSafeInteger(value) && value >= 0;

  const id = value =>
    typeof value === 'string' &&
    value.length > 0 &&
    !['__proto__', 'prototype', 'constructor']
      .includes(value);

  function clone(value) {
    if (typeof structuredClone === 'function') {
      return structuredClone(value);
    }

    return JSON.parse(JSON.stringify(value, (_, item) => {
      check(
        typeof item !== 'number' || Number.isFinite(item),
        '状态含非有限数值'
      );
      check(
        typeof item !== 'function',
        '函数只能放在配置，不能放在存档'
      );
      return item;
    }));
  }

  function phaseAt(time, config) {
    const day = Math.floor(time / 1440);
    const local = time % 1440;

    let index = config.phases.length - 1;

    while (config.phases[index].start > local) {
      index--;
    }

    const phase = config.phases[index];

    return {
      id: phase.id,
      start: day * 1440 + phase.start,
      end: day * 1440 +
        (config.phases[index + 1]?.start ?? 1440)
    };
  }

  function active(npc) {
    return npc.monitored === true &&
      npc.alive !== false;
  }

  function listIds(list, npcs, label) {
    check(Array.isArray(list), label + '必须是数组');

    list.forEach(value => {
      check(
        id(value) && own(npcs, value),
        label + '含未知角色'
      );
    });

    return new Set(list);
  }

  function unique(items, label) {
    const seen = new Set();

    for (const item of items || []) {
      check(
        id(item.id) && !seen.has(item.id),
        label + ' ID 缺失或重复'
      );
      seen.add(item.id);
    }
  }

  function validateConfig(config) {
    check(
      config &&
      Array.isArray(config.phases) &&
      config.phases[0]?.start === 0,
      '时段必须从每日第 0 分钟开始'
    );

    unique(config.phases, '时段');

    config.phases.forEach((p, i) => {
      check(
        minute(p.start) &&
        p.start < 1440 &&
        (!i || p.start > config.phases[i - 1].start),
        '时段必须递增且在一天以内'
      );
    });

    check(
      Number.isSafeInteger(
        config.policy?.ordinaryGraceMin
      ) &&
      config.policy.ordinaryGraceMin > 0,
      '请配置正整数普通活动保护时长 ordinaryGraceMin'
    );

    check(
      config.maxSteps === undefined ||
      (
        Number.isSafeInteger(config.maxSteps) &&
        config.maxSteps > 0
      ),
      'maxSteps 必须是正整数'
    );

    for (const name of ['actions', 'events', 'nodes']) {
      check(
        config[name] === undefined ||
        Array.isArray(config[name]),
        name + '必须是数组'
      );
      unique(config[name], name);
    }

    for (const event of [
      ...(config.events || []),
      ...(config.nodes || [])
    ]) {
      check(
        ['result', 'interaction'].includes(event.mode) &&
        typeof event.text === 'string' &&
        event.text.trim(),
        '事件或节点必须配置模式和文本'
      );

      check(
        !event.scope ||
        ['npc', 'world'].includes(event.scope),
        '事件 scope 非法'
      );
    }

    for (const node of config.nodes || []) {
      check(
        minute(node.fromStage) &&
        minute(node.toStage) &&
        node.toStage > node.fromStage,
        '节点阶段必须向前推进'
      );

      if (node.mode === 'interaction') {
        check(
          typeof node.completionText === 'string' &&
          node.completionText.trim(),
          '待互动节点必须配置 completionText'
        );
      }
    }

    const nodeIds = new Set(
      (config.nodes || []).map(n => n.id)
    );

    for (const node of config.nodes || []) {
      for (const required of node.requires || []) {
        check(
          nodeIds.has(required) &&
          required !== node.id,
          '节点前置引用非法'
        );
      }
    }

    const visiting = new Set();
    const visited = new Set();

    function visit(node) {
      check(
        !visiting.has(node.id),
        '节点前置条件存在循环'
      );

      if (visited.has(node.id)) return;

      visiting.add(node.id);

      for (const required of node.requires || []) {
        visit(
          config.nodes.find(item => item.id === required)
        );
      }

      visiting.delete(node.id);
      visited.add(node.id);
    }

    for (const node of config.nodes || []) {
      visit(node);
    }
  }

  function validateWorld(world) {
    check(
      minute(world.timeMin) &&
      id(world.playerLocation) &&
      world.npcs,
      '世界状态缺少时间、玩家位置或角色表'
    );

    for (const [key, npc] of
      Object.entries(world.npcs)) {
      check(
        id(key) &&
        npc.id === key &&
        id(npc.location),
        '角色 ID 或位置非法'
      );

      check(
        npc.stage === undefined || minute(npc.stage),
        '阶段非法'
      );

      if (npc.activity) {
        check(
          id(npc.activity.id) &&
          minute(npc.activity.startedAt) &&
          npc.activity.startedAt <= world.timeMin &&
          ['ordinary', 'important', 'special']
            .includes(npc.activity.kind),
          '角色活动存档非法'
        );
      }

      for (const [name, value] of
        Object.entries(npc.values || {})) {
        check(
          id(name) && Number.isFinite(value),
          '角色专属数值非法'
        );
      }
    }
  }

  function sortedNpcs(world) {
    return Object.values(world.npcs)
      .sort((a, b) => a.id.localeCompare(b.id));
  }

  function stable(value) {
    if (Array.isArray(value)) {
      return value.map(stable);
    }

    if (value && typeof value === 'object') {
      return Object.fromEntries(
        Object.keys(value)
          .sort()
          .map(name => [name, stable(value[name])])
      );
    }

    return value;
  }

  registry.utils = Object.freeze({
    own,
    check,
    minute,
    id,
    clone,
    phaseAt,
    active,
    listIds,
    validateConfig,
    validateWorld,
    sortedNpcs,
    key: (...parts) => JSON.stringify(parts),
    signature: value => JSON.stringify(stable(value))
  });

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = registry.utils;
  }
})(globalThis);

/**
 * @module okh_rules
 * @updated 2026-11-09
 * @status implemented
 * @reason 无副作用条件判断；缺少专属数值不视为零
 */
(function (root) {
  'use strict';

  function create({ utils: u }) {
    function compare(actual, expected) {
      if (
        !expected ||
        typeof expected !== 'object' ||
        Array.isArray(expected)
      ) {
        return actual === expected;
      }

      const operations = {
        eq: value => actual === value,
        ne: value => actual !== value,
        gt: value => actual > value,
        gte: value => actual >= value,
        lt: value => actual < value,
        lte: value => actual <= value
      };

      return Object.entries(expected)
        .every(([op, value]) => {
          u.check(
            u.own(operations, op),
            '未知比较运算：' + op
          );

          return actual !== undefined &&
            operations[op](value);
        });
    }

    function fields(actual, expected) {
      return Object.entries(expected || {})
        .every(([name, value]) =>
          actual &&
          u.own(actual, name) &&
          compare(actual[name], value)
        );
    }

    function matches(condition, ctx) {
      if (condition === undefined) return true;

      if (typeof condition === 'function') {
        return Boolean(condition(ctx));
      }

      u.check(
        condition && typeof condition === 'object',
        '条件必须是对象或可信配置函数'
      );

      const tests = {
        all: list =>
          list.every(item => matches(item, ctx)),

        any: list =>
          list.some(item => matches(item, ctx)),

        not: item => !matches(item, ctx),

        metrics: value =>
          fields(ctx.npc?.values, value),

        flags: value =>
          fields(ctx.npc?.flags, value),

        worldFlags: value =>
          fields(ctx.world.flags, value),

        time: value =>
          compare(ctx.world.timeMin, value),

        stage: value =>
          compare(ctx.npc?.stage, value),

        location: value =>
          (Array.isArray(value) ? value : [value])
            .includes(ctx.npc?.location),

        phase: value =>
          (Array.isArray(value) ? value : [value])
            .includes(ctx.phase.id),

        idle: value =>
          Boolean(
            !ctx.npc?.activity &&
            !ctx.blockedIds?.has(ctx.npc?.id)
          ) === value
      };

      return Object.entries(condition)
        .every(([name, value]) => {
          u.check(
            u.own(tests, name),
            '未知条件字段：' + name
          );
          return tests[name](value);
        });
    }

    return { matches, compare };
  }

  const registry =
    root.OKHBackgroundModules ||= Object.create(null);

  if (registry.rules) {
    throw Error('重复加载 okh_rules');
  }

  registry.rules = create;

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = create;
  }
})(globalThis);

/**
 * @module okh_activity
 * @updated 2026-11-09
 * @status implemented
 * @reason 管理活动占用、普通保护期及重要活动解除
 */
(function (root) {
  'use strict';

  function create({ utils: u, rules }) {
    function finish(npc, now, reason) {
      if (!npc.activity) return false;

      npc.activityHistory ||= {};

      npc.activityHistory[npc.activity.id] = {
        endedAt: now,
        reason
      };

      npc.activity = null;
      return true;
    }

    function start(npc, input, now, config) {
      u.check(
        u.id(input.id) &&
        ['ordinary', 'important', 'special']
          .includes(input.kind),
        '活动 ID 或类型非法'
      );

      u.check(
        !u.own(input, 'endsAt') &&
        !u.own(input, 'durationMin'),
        '保护期由统一配置管理，不接受逐活动耗时'
      );

      // 同一活动重复申报，不刷新计时。
      // 已结束的同一活动，也不能重新启动。
      if (
        npc.activity?.id === input.id ||
        u.own(npc.activityHistory || {}, input.id)
      ) {
        return false;
      }

      u.check(
        !npc.activity || input.replace === true,
        '角色已有活动，不能隐式覆盖'
      );

      if (
        input.kind !== 'ordinary' &&
        input.releaseRule !== undefined
      ) {
        u.check(
          u.id(input.releaseRule) &&
          u.own(
            config.releaseRules || {},
            input.releaseRule
          ),
          '活动解除规则不存在'
        );
      }

      u.check(
        input.kind !== 'ordinary' ||
        input.releaseRule === undefined,
        '普通活动不使用单独解除规则'
      );

      if (npc.activity) {
        finish(npc, now, 'explicit-interruption');
      }

      npc.activity = {
        id: input.id,
        kind: input.kind,
        label: input.label || input.kind,
        source: input.source || 'narrative',
        startedAt: now
      };

      if (input.releaseRule !== undefined) {
        npc.activity.releaseRule = input.releaseRule;
      }

      return true;
    }

    function change(npc, input, now, config) {
      u.check(
        input &&
        u.id(input.id) &&
        ['start', 'finish', 'cancel'].includes(input.op),
        '活动变更 ID 或 op 非法'
      );

      if (input.op === 'start') {
        return start(npc, input, now, config);
      }

      if (
        u.own(npc.activityHistory || {}, input.id)
      ) {
        return false;
      }

      u.check(
        npc.activity?.id === input.id,
        '结束活动的 ID 与当前活动不一致'
      );

      return finish(
        npc,
        now,
        input.op === 'cancel'
          ? 'explicit-cancel'
          : 'explicit-finish'
      );
    }

    function refresh(world, config, ctx) {
      const released = [];

      for (const npc of u.sortedNpcs(world)) {
        if (!u.active(npc)) continue;

        const task = npc.activity;
        if (!task) continue;

        if (
          task.kind !== 'ordinary' &&
          task.releaseRule !== undefined
        ) {
          u.check(
            u.own(
              config.releaseRules || {},
              task.releaseRule
            ),
            '活动解除规则被删除，不能自动释放占用'
          );
        }

        const expired =
          task.kind === 'ordinary' &&
          world.timeMin - task.startedAt >=
            config.policy.ordinaryGraceMin;

        const conditionMet =
          task.kind !== 'ordinary' &&
          task.releaseRule !== undefined &&
          rules.matches(
            config.releaseRules[task.releaseRule],
            { ...ctx, world, npc }
          );

        if (expired || conditionMet) {
          finish(
            npc,
            world.timeMin,
            expired
              ? 'ordinary-timeout'
              : 'release-condition'
          );

          released.push(npc.id);
        }
      }

      return released;
    }

    function nextDeadline(world, config) {
      let next = Infinity;

      for (const npc of Object.values(world.npcs)) {
        if (
          !u.active(npc) ||
          npc.activity?.kind !== 'ordinary'
        ) {
          continue;
        }

        const deadline =
          npc.activity.startedAt +
          config.policy.ordinaryGraceMin;

        u.check(
          u.minute(deadline),
          '活动保护期溢出'
        );

        if (deadline > world.timeMin) {
          next = Math.min(next, deadline);
        }
      }

      return next;
    }

    function status(npc, interactingIds) {
      if (!u.active(npc)) return 'inactive';

      if (interactingIds.has(npc.id)) {
        return 'interacting';
      }

      return npc.activity?.kind || 'idle';
    }

    return {
      start,
      change,
      finish,
      refresh,
      nextDeadline,
      status
    };
  }

  const registry =
    root.OKHBackgroundModules ||= Object.create(null);

  if (registry.activity) {
    throw Error('重复加载 okh_activity');
  }

  registry.activity = create;

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = create;
  }
})(globalThis);

/**
 * @module okh_effects
 * @updated 2026-11-09
 * @status implemented
 * @reason 集中结算位置、专属数值和标记
 */
(function (root) {
  'use strict';

  function create({ utils: u }) {
    function one(world, npc, data, ctx, config) {
      const allowed = new Set([
        'move',
        'add',
        'npcFlags',
        'worldFlags'
      ]);

      Object.keys(data).forEach(name => {
        u.check(
          allowed.has(name),
          '未知效果字段：' + name
        );
      });

      u.check(
        npc ||
        (!data.add && !data.move && !data.npcFlags),
        '世界效果不能隐式修改角色'
      );

      if (data.move !== undefined) {
        u.check(
          u.id(data.move) &&
          !ctx.blockedIds.has(npc.id) &&
          !npc.activity,
          '不能后台移动互动中或被活动占用的角色'
        );

        npc.location = data.move;
      }

      for (const [name, delta] of
        Object.entries(data.add || {})) {
        u.check(
          npc.values && u.own(npc.values, name),
          '角色未声明专属数值：' + name
        );

        let value = npc.values[name] + delta;

        u.check(
          Number.isFinite(delta) &&
          Number.isFinite(value),
          '数值变化非法'
        );

        const bounds =
          config.metricBounds?.[npc.id]?.[name];

        if (bounds) {
          u.check(
            bounds.length === 2 &&
            bounds.every(Number.isFinite) &&
            bounds[0] <= bounds[1],
            '数值边界非法'
          );

          value = Math.max(
            bounds[0],
            Math.min(bounds[1], value)
          );
        }

        npc.values[name] = value;
      }

      if (npc) {
        npc.flags = {
          ...npc.flags,
          ...data.npcFlags
        };
      }

      world.flags = {
        ...world.flags,
        ...data.worldFlags
      };
    }

    function apply(world, actor, effect, ctx, config) {
      if (!effect) return;

      if (Array.isArray(effect)) {
        for (const item of effect) {
          u.check(
            u.own(item, 'target'),
            '跨目标效果必须显式指定 target'
          );

          const { target, ...data } = item;

          u.check(
            target === 'world' ||
            u.own(world.npcs, target),
            '效果目标不存在'
          );

          one(
            world,
            target === 'world'
              ? null
              : world.npcs[target],
            data,
            ctx,
            config
          );
        }
      } else {
        one(world, actor, effect, ctx, config);
      }
    }

    return { apply };
  }

  const registry =
    root.OKHBackgroundModules ||= Object.create(null);

  if (registry.effects) {
    throw Error('重复加载 okh_effects');
  }

  registry.effects = create;

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = create;
  }
})(globalThis);

/**
 * @module okh_events
 * @updated 2026-11-09
 * @status implemented
 * @reason 条件事件和演绎队列，不访问HUD或调用LLM
 */
(function (root) {
  'use strict';

  function create({ utils: u, rules, effects }) {
    function emit(world, entry, npc) {
      if (
        world.bg.queue.some(item => item.id === entry.id)
      ) {
        return false;
      }

      const fields = {
        name: npc?.name || npc?.id || '',
        location: npc?.location || ''
      };

      world.bg.queue.push({
        ...entry,
        at: world.timeMin,
        npcId: npc?.id ?? null,
        text: entry.text.replace(
          /\{(name|location)\}/g,
          (_, name) => fields[name]
        ),
        status: 'pending'
      });

      return true;
    }

    function evaluate(world, config, ctx) {
      for (const event of config.events || []) {
        const targets = event.scope === 'world'
          ? [null]
          : u.sortedNpcs(world).filter(u.active);

        for (const npc of targets) {
          if (
            npc &&
            event.targets &&
            !event.targets.includes(npc.id)
          ) {
            continue;
          }

          const key = u.key(
            'event',
            event.id,
            npc?.id ?? null
          );

          if (world.bg.fired[key]) continue;

          if (
            npc &&
            event.idleOnly &&
            (
              npc.activity ||
              ctx.blockedIds.has(npc.id)
            )
          ) {
            continue;
          }

          if (
            !rules.matches(
              event.when,
              { ...ctx, world, npc }
            )
          ) {
            continue;
          }

          effects.apply(
            world,
            npc,
            event.effects,
            ctx,
            config
          );

          world.bg.fired[key] = true;

          emit(world, {
            id: key,
            definitionId: event.id,
            kind: 'event',
            mode: event.mode,
            text: event.text
          }, npc);
        }
      }
    }

    function prepare(world, limit = 8) {
      u.check(
        Number.isSafeInteger(limit) && limit > 0,
        '批次大小非法'
      );

      const batch = world.bg.queue
        .filter(e => e.status === 'pending')
        .slice(0, limit);

      return {
        ids: batch.map(e => e.id),

        text: batch.map(e => {
          const instruction = e.mode === 'result'
            ? '【已定事实：保持结果并演绎】'
            : '【待互动：只展开局面，不替玩家决定；未决结果须另行结算】';

          return instruction + '\n' + e.text;
        }).join('\n\n')
      };
    }

    function acknowledge(world, ids, generationId) {
      u.check(
        u.id(generationId) && Array.isArray(ids),
        '确认生成 ID 或事件批次非法'
      );

      const selected = new Set(ids);

      for (const id of selected) {
        u.check(
          world.bg.queue.some(e => e.id === id),
          '待确认事件不存在'
        );
      }

      for (const event of world.bg.queue) {
        if (
          selected.has(event.id) &&
          event.status === 'pending'
        ) {
          event.status = 'delivered';
          event.generationId = generationId;
        }
      }
    }

    function supersede(world, id) {
      const entry = world.bg.queue
        .find(e => e.id === id);

      if (entry?.status === 'pending') {
        entry.status = 'superseded';
      }
    }

    return {
      evaluate,
      emit,
      prepare,
      acknowledge,
      supersede
    };
  }

  const registry =
    root.OKHBackgroundModules ||= Object.create(null);

  if (registry.events) {
    throw Error('重复加载 okh_events');
  }

  registry.events = create;

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = create;
  }
})(globalThis);

/**
 * @module okh_nodes
 * @updated 2026-11-09
 * @status implemented
 * @reason 专属数值经节点推进阶段
 */
(function (root) {
  'use strict';

  function create({ utils: u, rules, effects, events }) {
    const queueId = (node, npc) =>
      u.key('node', node.id, npc.id);

    function complete(
      world,
      npc,
      node,
      config,
      ctx,
      fromPending
    ) {
      u.check(
        npc.stage === node.fromStage,
        '节点完成时阶段已变化，需要处理冲突'
      );

      effects.apply(
        world,
        npc,
        node.effects,
        ctx,
        config
      );

      npc.stage = node.toStage;

      npc.nodeStates[node.id] = {
        ...npc.nodeStates[node.id],
        status: 'completed',
        completedAt: world.timeMin
      };

      if (fromPending) {
        events.supersede(world, queueId(node, npc));
      }

      events.emit(world, {
        id: fromPending
          ? u.key('node-result', node.id, npc.id)
          : queueId(node, npc),
        kind: 'node',
        definitionId: node.id,
        mode: 'result',
        text: fromPending
          ? node.completionText
          : node.text
      }, npc);
    }

    function evaluate(world, config, ctx) {
      const definitions = [...(config.nodes || [])]
        .sort((a, b) =>
          (b.priority || 0) - (a.priority || 0) ||
          a.id.localeCompare(b.id)
        );

      for (const npc of
        u.sortedNpcs(world).filter(u.active)) {
        npc.nodeStates ||= {};

        if (
          Object.values(npc.nodeStates)
            .some(record => record.status === 'pending')
        ) {
          continue;
        }

        const node = definitions.find(item =>
          (npc.nodePlan || []).includes(item.id) &&
          !u.own(npc.nodeStates, item.id) &&
          npc.stage === item.fromStage &&
          (item.requires || []).every(
            id =>
              npc.nodeStates[id]?.status === 'completed'
          ) &&
          rules.matches(
            item.when,
            { ...ctx, world, npc }
          )
        );

        if (!node) continue;

        npc.nodeStates[node.id] = {
          status: 'pending',
          triggeredAt: world.timeMin
        };

        if (node.mode === 'result') {
          complete(
            world,
            npc,
            node,
            config,
            ctx,
            false
          );
        } else {
          events.emit(world, {
            id: queueId(node, npc),
            kind: 'node',
            definitionId: node.id,
            mode: 'interaction',
            text: node.text
          }, npc);
        }
      }
    }

    function resolve(world, resolutions, config, ctx) {
      for (const resolution of resolutions || []) {
        const npc = world.npcs[resolution.npcId];

        const node = (config.nodes || [])
          .find(n => n.id === resolution.nodeId);

        u.check(
          npc &&
          node &&
          (npc.nodePlan || []).includes(node.id),
          '待处理节点或角色不存在'
        );

        const record = npc.nodeStates?.[node.id];

        u.check(record, '节点尚未触发');

        u.check(
          ['complete', 'cancel']
            .includes(resolution.outcome),
          '节点处理结果非法'
        );

        if (
          record.status === 'completed' &&
          resolution.outcome === 'complete'
        ) {
          continue;
        }

        if (
          record.status === 'cancelled' &&
          resolution.outcome === 'cancel'
        ) {
          continue;
        }

        u.check(
          record.status === 'pending',
          '节点已经处理，不能变更结果'
        );

        if (resolution.outcome === 'cancel') {
          record.status = 'cancelled';
          record.cancelledAt = world.timeMin;

          events.supersede(
            world,
            queueId(node, npc)
          );
        } else {
          complete(
            world,
            npc,
            node,
            config,
            ctx,
            true
          );
        }
      }
    }

    return { evaluate, resolve };
  }

  const registry =
    root.OKHBackgroundModules ||= Object.create(null);

  if (registry.nodes) {
    throw Error('重复加载 okh_nodes');
  }

  registry.nodes = create;

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = create;
  }
})(globalThis);

(function (root) {
  'use strict';

  function create({
    utils: u,
    rules,
    activity,
    effects,
    events,
    nodes
  }) {
    function refresh(
      world,
      config,
      blockedIds,
      report
    ) {
      const ctx = {
        world,
        phase: u.phaseAt(world.timeMin, config),
        blockedIds
      };

      report.released.push(
        ...activity.refresh(world, config, ctx)
      );

      events.evaluate(world, config, ctx);
      nodes.evaluate(world, config, ctx);

      report.released.push(
        ...activity.refresh(world, config, ctx)
      );
    }

    function schedule(
      world,
      config,
      blockedIds,
      report
    ) {
      const phase = u.phaseAt(world.timeMin, config);
      const snapshot = u.clone(world);
      const decisions = [];

      for (const npc of u.sortedNpcs(snapshot)) {
        if (
          activity.status(npc, blockedIds) !== 'idle' ||
          world.bg.settled[npc.id] === phase.start
        ) {
          continue;
        }

        const ctx = {
          world: snapshot,
          npc,
          phase,
          blockedIds
        };

        const action = (config.actions || [])
          .filter(item =>
            (npc.plan || []).includes(item.id) &&
            rules.matches(item.when, ctx) &&
            (
              !config.allowed ||
              config.allowed(ctx, item)
            )
          )
          .sort((a, b) =>
            (b.priority || 0) - (a.priority || 0) ||
            a.id.localeCompare(b.id)
          )[0];

        decisions.push({
          id: npc.id,
          action
        });
      }

      for (const { id, action } of decisions) {
        world.bg.settled[id] = phase.start;

        if (!action) continue;

        const npc = world.npcs[id];

        if (
          activity.status(npc, blockedIds) !== 'idle'
        ) {
          continue;
        }

        const ctx = {
          world,
          npc,
          phase,
          blockedIds
        };

        if (
          !rules.matches(action.when, ctx) ||
          (
            config.allowed &&
            !config.allowed(ctx, action)
          )
        ) {
          continue;
        }

        effects.apply(
          world,
          npc,
          action.effects,
          ctx,
          config
        );

        if (action.occupation) {
          activity.start(npc, {
            ...action.occupation,
            id: u.key(
              'action',
              action.id,
              id,
              world.timeMin
            ),
            source: 'background'
          }, world.timeMin, config);
        }

        npc.lastAction = {
          id: action.id,
          at: world.timeMin
        };

        report.actions.push({
          npcId: id,
          actionId: action.id,
          at: world.timeMin
        });
      }
    }

    function applyFront(
      world,
      front,
      config,
      blockedIds
    ) {
      if (front.playerLocation !== undefined) {
        u.check(
          u.id(front.playerLocation),
          '玩家位置非法'
        );

        world.playerLocation = front.playerLocation;
      }

      for (const [id, patch] of
        Object.entries(front.npcs || {})) {
        u.check(
          u.own(world.npcs, id),
          '前台回执角色不存在'
        );

        u.check(
          !u.own(patch, 'stage'),
          '阶段只能由节点推进，不能由前台直接覆盖'
        );

        const npc = world.npcs[id];

        if (patch.location !== undefined) {
          u.check(
            u.id(patch.location),
            '角色位置非法'
          );

          npc.location = patch.location;
        }

        effects.apply(world, npc, {
          add: patch.add,
          npcFlags: patch.flags
        }, { blockedIds }, config);

        if (patch.activity) {
          activity.change(
            npc,
            patch.activity,
            world.timeMin,
            config
          );
        }
      }

      world.flags = {
        ...world.flags,
        ...front.flags
      };

      nodes.resolve(
        world,
        front.resolveNodes,
        config,
        {
          world,
          phase: u.phaseAt(world.timeMin, config),
          blockedIds
        }
      );
    }

    function advance(world, command, config) {
      u.validateConfig(config);
      u.validateWorld(world);

      u.check(
        world.bg?.version === 2,
        '请先初始化，旧版状态须显式迁移'
      );

      u.check(
        u.id(command.id) && u.id(command.revision),
        '需要稳定聊天消息 ID 和回复版本'
      );

      const signature = u.signature({
        revision: command.revision,
        toMin: command.toMin,
        interactingIds:
          command.interactingIds ??
          command.sceneIds ??
          [],
        front: command.front ?? null
      });

      const previous =
        world.bg.receipts[command.id];

      if (previous !== undefined) {
        u.check(
          previous === signature,
          'REPLAY_REQUIRED：先恢复该消息之前的检查点'
        );

        return {
          state: u.clone(world),
          before: null,
          report: {
            duplicate: true,
            released: [],
            actions: []
          },
          duplicate: true
        };
      }

      u.check(
        u.minute(command.toMin) &&
        command.toMin >= world.timeMin,
        '时间倒退须恢复检查点，不能直接减时钟'
      );

      const state = u.clone(world);
      const front = command.front || {};

      // sceneIds仅兼容旧接口。
      // 它现在表示互动角色，不再由位置推导。
      const initialIds =
        command.interactingIds ??
        command.sceneIds ??
        state.interactingIds ??
        [];

      const blockedIds = u.listIds(
        initialIds,
        state.npcs,
        '互动角色列表'
      );

      state.interactingIds = [...blockedIds];

      // 新报告活动先做保护，防止回补时间时被抢占。
      for (const [id, patch] of
        Object.entries(front.npcs || {})) {
        u.check(
          u.own(state.npcs, id),
          '前台回执角色不存在'
        );

        const task = patch.activity;
        const npc = state.npcs[id];

        if (
          task?.op === 'start' &&
          npc.activity?.id !== task.id &&
          !u.own(
            npc.activityHistory || {},
            task.id
          )
        ) {
          blockedIds.add(id);
        }
      }

      const report = {
        duplicate: false,
        released: [],
        actions: []
      };

      refresh(state, config, blockedIds, report);

      let steps = 0;

      while (state.timeMin < command.toMin) {
        u.check(
          ++steps <= (config.maxSteps ?? 128),
          '时间推进过多，本次未提交任何变化'
        );

        const boundary =
          u.phaseAt(state.timeMin, config).end;

        const next = Math.min(
          command.toMin,
          boundary,
          activity.nextDeadline(state, config)
        );

        state.timeMin = next;

        refresh(state, config, blockedIds, report);

        if (next === boundary) {
          schedule(state, config, blockedIds, report);
          refresh(state, config, blockedIds, report);
        }
      }

      applyFront(state, front, config, blockedIds);

      if (front.interactingIds !== undefined) {
        state.interactingIds = [
          ...u.listIds(
            front.interactingIds,
            state.npcs,
            '更新后的互动角色列表'
          )
        ];
      }

      refresh(
        state,
        config,
        new Set(state.interactingIds),
        report
      );

      state.bg.receipts[command.id] = signature;
      state.bg.revision++;

      u.validateWorld(state);

      return {
        state,
        before: u.clone(world),
        report,
        duplicate: false
      };
    }

    return { advance };
  }

  const registry =
    root.OKHBackgroundModules ||= Object.create(null);

  if (registry.turn) {
    throw Error('重复加载 okh_turn');
  }

  registry.turn = create;

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = create;
  }
})(globalThis);

 * @module okh_adapter

(function (root) {
  'use strict';

  const u = root.OKHBackgroundModules?.utils;
  const api = root.OKHBackground;

  if (!u || !api) {
    throw Error(
      'okh_adapter必须在okh_api后加载'
    );
  }

  function createController({
    readWorld,
    commit,
    config
  }) {
    u.check(
      typeof readWorld === 'function' &&
      typeof commit === 'function',
      '需要提供现有状态读写适配器'
    );

    u.validateConfig(config);

    let pending = Promise.resolve();

    function serial(task) {
      const result = pending.then(task);
      pending = result.catch(() => {});
      return result;
    }

    async function save(
      before,
      state,
      checkpoint,
      reason
    ) {
      const result = await commit({
        expectedRevision: before.bg.revision,
        state,
        checkpoint,
        reason
      });

      u.check(
        result === true,
        '保存冲突或失败，未确认提交成功'
      );
    }

    return {
      prepareGeneration(generationId, limit = 8) {
        return serial(async () => {
          u.check(
            u.id(generationId),
            '生成ID非法'
          );

          const before = await readWorld();

          u.check(
            before.bg?.version === 2,
            '需要初始化后的v2状态'
          );

          const previous =
            before.bg.generations[generationId];

          if (previous) {
            u.check(
              previous.status === 'prepared',
              '本生成请求已经完成'
            );

            const view = u.clone(before);

            view.bg.queue = view.bg.queue.filter(
              entry => previous.ids.includes(entry.id)
            );

            const batch = api.prepare(
              view,
              Math.max(previous.ids.length, 1)
            );

            u.check(
              u.signature(batch.ids) ===
                u.signature(previous.ids),
              '已准备事件发生变化，请使用新生成ID'
            );

            return batch;
          }

          const state = u.clone(before);
          const batch = api.prepare(state, limit);

          state.bg.generations[generationId] = {
            ids: batch.ids,
            status: 'prepared'
          };

          state.bg.revision++;

          await save(
            before,
            state,
            null,
            'prepare-generation'
          );

          return batch;
        });
      },

      finishReply(command, generationId = null) {
        return serial(async () => {
          const before = await readWorld();

          const result = api.advance(
            before,
            command,
            config
          );

          if (result.duplicate) return result;

          if (generationId !== null) {
            const generation =
              before.bg.generations[generationId];

            u.check(
              generation?.status === 'prepared',
              '回复没有对应的待完成生成请求'
            );

            result.state = api.ack(
              result.state,
              generation.ids,
              generationId
            );

            result.state.bg
              .generations[generationId]
              .status = 'completed';

            result.state.bg
              .generations[generationId]
              .replyId = command.id;
          }

          await save(
            before,
            result.state,
            before,
            'finish-reply'
          );

          return result;
        });
      },

      restore(checkpoint) {
        return serial(async () => {
          u.check(
            checkpoint?.bg?.version === 2,
            '检查点必须是完整v2状态'
          );

          u.validateWorld(checkpoint);

          const before = await readWorld();
          const state = u.clone(checkpoint);

          // 状态回退，但提交版本仍递增。
          state.bg.revision =
            before.bg.revision + 1;

          await save(
            before,
            state,
            null,
            'restore-checkpoint'
          );

          return state;
        });
      }
    };
  }

  if (root.OKHBackgroundAdapter) {
    throw Error('重复加载 okh_adapter');
  }

    createController
  });

  }

                 : op === 'eq' ? v === n
                 : op === 'ne' ? v !== n

    }
    return true

/**
 * @module okh_app_world
 * @updated 2026-11-09 06:45
 * @status  demo
 * @reason  world 的读写与首次初始化。存储适配层让同一份代码在酒馆和 Node 都能跑
 * @deps    ['okh_app_config'] + gpt_core_all（初始化时需要 OKHBackground）
 * @provides { store, loadWorld, saveWorld, ensureWorld, resetWorld, STORAGE_KEY }
 */

(function (root) {
  'use strict'

  const CFG = root.OKH_APP_CONFIG
  if (!CFG) { console.error('[APP] 需要先加载 01_config.js'); return }

  const STORAGE_KEY = 'okh_gpt.world'

  // ============================================================
  // 一、存储适配层
  //    酒馆里用 getvar/setvar，Node 里退回内存 Map
  // ============================================================

  const store = (function () {
    if (typeof getvar === 'function' && typeof setvar === 'function') {
      return {
        mode: 'tavern',
        get: k => { try { return getvar(k) } catch (e) { return undefined } },
        set: (k, v) => { try { setvar(k, v) } catch (e) {} },
        del: k => { try { if (typeof deletevar === 'function') deletevar(k) } catch (e) {} }
      }
    }
    const mem = new Map()
    return {
      mode: 'memory',
      get: k => mem.get(k),
      set: (k, v) => { mem.set(k, v) },
      del: k => { mem.delete(k) }
    }
  })()

  // ============================================================
  // 二、读写
  // ============================================================

  /**
   * 读 world
   * @returns {object|null}
   */
  function loadWorld() {
    const raw = store.get(STORAGE_KEY)
    if (raw === undefined || raw === null) return null
    if (typeof raw === 'string') {
      try { return JSON.parse(raw) } catch (e) { return null }
    }
    return raw
  }

  /**
   * 写 world
   * @param {object} world
   * @returns {boolean} 回读校验结果
   */
  function saveWorld(world) {
    const text = JSON.stringify(world)
    store.set(STORAGE_KEY, text)

    const back = store.get(STORAGE_KEY)
    const ok = back === text
    if (!ok) console.warn('[APP] world 写入校验失败')
    return ok
  }

  // ============================================================
  // 三、初始化
  // ============================================================

  /**
   * 确保 world 存在；不存在就初始化
   * @param {boolean} force 为 true 时强制重建
   * @returns {object}
   */
  function ensureWorld(force = false) {
    if (!force) {
      const existing = loadWorld()
      if (existing && existing.bg?.version === 2) return existing
    }

    const GPT = root.OKHBackground
    if (!GPT) {
      throw new Error('[APP] 需要先加载 gpt_core_all.js')
    }

    const world = GPT.init(CFG.buildInitialWorld(), CFG.buildConfig())
    saveWorld(world)
    console.log('[APP] world 已初始化 · ' + Object.keys(world.npcs).length + ' 个角色')
    return world
  }

  /**
   * 清空重来
   * @returns {object}
   */
  function resetWorld() {
    store.del(STORAGE_KEY)
    return ensureWorld(true)
  }

  root.OKH_APP_WORLD = Object.freeze({
    STORAGE_KEY,
    store,
    loadWorld,
    saveWorld,
    ensureWorld,
    resetWorld
  })

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = root.OKH_APP_WORLD
  }
})(typeof globalThis !== 'undefined' ? globalThis : this)

/**
 * @module okh_app_adapter
 * @updated 2026-11-09 06:50
 * @status  demo
 * @reason  用 GPT 提供的 createController 接上宿主读写；额外提供驱动漂移与台账生成
 * @deps    ['okh_app_config', 'okh_app_world'] + gpt_core_all
 * @provides { createController, drift, buildLedger, buildWatch }
 */

(function (root) {
  'use strict'

  const CFG = root.OKH_APP_CONFIG
  const W = root.OKH_APP_WORLD
  const GPT = root.OKHBackground
  const GPT_ADAPTER = root.OKHBackgroundAdapter

  if (!CFG || !W) { console.error('[APP] 需要先加载 01_config 与 02_world'); return }
  if (!GPT || !GPT_ADAPTER) { console.error('[APP] 需要先加载 gpt_core_all.js'); return }

  // ============================================================
  // 一、控制器
  // ============================================================

  /**
   * 建立 GPT 的控制器
   * readWorld / commit 都指向同一份真源（变量池里的 world）
   * @returns {object}
   */
  function createController() {
    const config = CFG.buildConfig()

    return GPT_ADAPTER.createController({
      config,

      async readWorld() {
        const w = W.loadWorld()
        if (!w) return W.ensureWorld()
        return w
      },

      async commit({ expectedRevision, state, checkpoint, reason }) {
        const current = W.loadWorld()

        // 版本冲突检查：别人先改了，就不要覆盖
        if (current && current.bg && current.bg.revision !== expectedRevision) {
          console.warn('[APP] 版本冲突：期望 ' + expectedRevision + '，实际 ' + current.bg.revision)
          return false
        }

        const ok = W.saveWorld(state)

        // 检查点与本次回复绑定保存（用于回滚）
        if (ok && checkpoint) {
          W.store.set('okh_gpt.checkpoint', JSON.stringify(checkpoint))
        }

        return ok
      }
    })
  }

  // ============================================================
  // 二、驱动力漂移（GPT 核心没有这个机制，在应用层补）
  // ============================================================

  /**
   * 每个时段结束：驱动力向基线回归 + 生活压力
   * 直接原地修改 world，调用方负责保存
   * @param {object} world
   * @param {object} opts { regression, needPressure }
   * @returns {number} 受影响角色数
   */
  function drift(world, opts = {}) {
    const regression = opts.regression ?? 0.10
    const needPressure = opts.needPressure ?? 2
    let touched = 0

    for (const npc of Object.values(world.npcs)) {
      if (npc.alive === false) continue
      const base = npc._baseline
      if (!base) continue

      for (const k of Object.keys(npc.values)) {
        if (k === 'need') continue          // need 单独加压力
        if (typeof base[k] !== 'number') continue
        const v = npc.values[k]
        npc.values[k] = Math.max(0, Math.min(100, v + (base[k] - v) * regression))
      }

      if (typeof npc.values.need === 'number') {
        npc.values.need = Math.max(0, Math.min(100, npc.values.need + needPressure))
      }

      touched++
    }

    return touched
  }

  // ============================================================
  // 三、台账与监听
  // ============================================================

  /**
   * 世界快照，给 AI 看
   * @param {object} world
   * @returns {string}
   */
  function buildLedger(world) {
    const day = Math.floor(world.timeMin / 1440) + 1
    const local = world.timeMin % 1440
    const phase = local < 360 ? '深夜'
                : local < 600 ? '上午'
                : local < 1020 ? '午后'
                : '夜晚'

    const lines = [`【第 ${day} 天 · ${phase}】`]

    // 在场：和玩家同位置的
    const here = Object.values(world.npcs)
      .filter(n => n.alive !== false && n.location === world.playerLocation)

    lines.push('在场：' + (here.length
      ? here.map(n => {
          const busy = n.activity ? `（${n.activity.label}）` : ''
          return n.name + busy
        }).join('、')
      : '无'))

    // 近况：队列里待演绎的
    const pending = (world.bg?.queue || [])
      .filter(e => e.status === 'pending')
      .slice(-3)

    if (pending.length) {
      lines.push('近况：')
      for (const e of pending) lines.push('· ' + e.text)
    }

    return lines.join('\n')
  }

  /**
   * 监听快照，不给 AI 看
   * @param {object} world
   * @returns {object}
   */
  function buildWatch(world) {
    const STAGE = { 0: '正常', 1: '异常', 2: '沉沦', 3: '臣服' }
    const list = Object.values(world.npcs).map(n => ({
      id: n.id,
      name: n.name,
      location: n.location,
      stage: STAGE[n.stage] || n.stage,
      corruption: Math.round(n.values.corruption ?? 0),
      sanity: Math.round(n.values.sanity ?? 0),
      exposure: Math.round(n.values.exposure ?? 0),
      activity: n.activity?.label || null,
      present: n.location === world.playerLocation
    }))

    return {
      timeMin: world.timeMin,
      day: Math.floor(world.timeMin / 1440) + 1,
      list
    }
  }

  root.OKH_APP_ADAPTER = Object.freeze({
    createController,
    drift,
    buildLedger,
    buildWatch
  })

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = root.OKH_APP_ADAPTER
  }
})(typeof globalThis !== 'undefined' ? globalThis : this)

/**
 * @module okh_app_parse
 * @updated 2026-11-09 08:00
 * @status  demo
 * @reason  解析 AI 输出里的申报标签，转成 GPT 的 front 结构
 * @deps    无（纯函数）
 * @provides { parseActivity, parseInteract, parseStatusBar, buildFront, VALID_PHASES }
 *
 * ============================================================
 * 支持的标签
 *
 *   <activity npc="aurora" op="start" kind="ordinary" label="吃饭" />
 *   <activity npc="aurora" op="finish" />
 *   <activity npc="cecilia" op="start" kind="important" label="照看孤儿" rule="assignmentResolved" />
 *   <interact npc="aurora" />
 *   <move to="manor" />
 *   <focus npc="aurora" />
 *
 * 说明：
 *   · activity 的 id 由脚本生成（npc + 时间戳），不需要 AI 提供
 *   · finish / cancel 不需要给 id，脚本会匹配该角色的当前活动
 *   · interact 表示该角色本轮与玩家互动，后台不应给他安排行动
 * ============================================================
 */

(function (root) {
  'use strict'

  /**
   * 从文本里抓所有自闭合标签，返回 { tagName: [attrs...] }
   * @param {string} text
   * @param {string} tagName
   * @returns {object[]}
   */
  function grabTags(text, tagName) {
    const re = new RegExp(`<${tagName}\\b([^>]*)/?>`, 'gi')
    const out = []
    let m
    while ((m = re.exec(text)) !== null) {
      out.push(parseAttrs(m[1]))
    }
    return out
  }

  /**
   * 解析属性串
   * @param {string} raw 形如 ` npc="aurora" op="start" `
   * @returns {object}
   */
  function parseAttrs(raw) {
    const attrs = {}
    const re = /([\w-]+)\s*=\s*"([^"]*)"/g
    let m
    while ((m = re.exec(raw)) !== null) {
      attrs[m[1]] = m[2]
    }
    return attrs
  }

  /**
   * 解析 <activity>，产出 each npc 的活动变更指令
   * @param {string} text
   * @param {number} now 当前世界分钟，用于生成活动 id
   * @param {object} world 当前世界，用于取当前活动 id
   * @returns {object} { npcId: { activity } }
   */
  function parseActivity(text, now, world) {
    const tags = grabTags(text, 'activity')
    const out = {}

    for (const a of tags) {
      const npc = a.npc
      const op = a.op
      if (!npc || !op) continue
      if (!['start', 'finish', 'cancel'].includes(op)) continue

      const cur = world?.npcs?.[npc]?.activity

      if (!out[npc]) out[npc] = {}

      if (op === 'start') {
        const kind = ['ordinary', 'important', 'special'].includes(a.kind) ? a.kind : 'ordinary'
        out[npc].activity = {
          op: 'start',
          id: `act-${npc}-${now}`,
          kind,
          label: a.label || kind,
          source: 'narrative'
        }
        if (a.rule && kind !== 'ordinary') {
          out[npc].activity.releaseRule = a.rule
        }
      } else {
        // finish / cancel 必须带上当前活动 id，否则 GPT 会拒绝
        if (!cur) continue
        out[npc].activity = { op, id: cur.id }
      }
    }

    return out
  }

  /**
   * 解析 <interact>，产出互动角色列表
   * @param {string} text
   * @returns {string[]}
   */
  function parseInteract(text) {
    return grabTags(text, 'interact')
      .map(a => a.npc)
      .filter(Boolean)
  }

  /**
   * 解析 <move to="...">
   * @param {string} text
   * @returns {string|null}
   */
  function parseMove(text) {
    const tags = grabTags(text, 'move')
    return tags[0]?.to || null
  }

  /**
   * 解析 <time elapsed="30" end="1日18:30" />
   * @param {string} text
   * @returns {{elapsed:number|null, end:string|null}}
   */
  function parseStatusBar(text) {
    const tags = grabTags(text, 'time')
    const t = tags[0]
    if (!t) return { elapsed: null, end: null }

    let elapsed = null
    if (t.elapsed !== undefined) {
      const n = Number(t.elapsed)
      if (Number.isFinite(n) && n >= 0 && n <= 1440) elapsed = n
    }
    return { elapsed, end: t.end || null }
  }

  /**
   * 把解析结果组装成 GPT 的 front
   * @param {string} text AI 消息全文
   * @param {number} now 当前世界分钟
   * @param {object} world 当前 world（用于校验 npc 是否存在）
   * @returns {object} front
   */
  function buildFront(text, now, world) {
    const front = {}

    // 玩家移动
    const move = parseMove(text)
    if (move) front.playerLocation = move

    // 角色活动
    const acts = parseActivity(text, now, world)
    const npcs = {}
    for (const [npcId, patch] of Object.entries(acts)) {
      if (!world.npcs[npcId]) continue      // 未知角色，丢弃
      npcs[npcId] = patch
    }
    if (Object.keys(npcs).length) front.npcs = npcs

    // 互动角色
    const interacting = parseInteract(text)
      .filter(id => world.npcs[id])
    if (interacting.length) {
      front.interactingIds = interacting
    }

    return front
  }

  root.OKH_APP_PARSE = Object.freeze({
    grabTags,
    parseAttrs,
    parseActivity,
    parseInteract,
    parseMove,
    parseStatusBar,
    buildFront
  })

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = root.OKH_APP_PARSE
  }
})(typeof globalThis !== 'undefined' ? globalThis : this)
;
/**
 * @module okh_app_hook
 * @updated 2026-11-09 06:55
 * @status  demo
 * @reason  消息钩子与节奏控制：定时推进、跨时段推进、写台账
 * @deps    ['okh_app_config', 'okh_app_world', 'okh_app_adapter'] + gpt_core_all
 * @provides { RHYTHM, attach, detach, onMessage, step, lastReport, pendingText }
 */

(function (root) {
  'use strict'

  const CFG = root.OKH_APP_CONFIG
  const W = root.OKH_APP_WORLD
  const A = root.OKH_APP_ADAPTER
  const GPT = root.OKHBackground

  if (!CFG || !W || !A || !GPT) {
    console.error('[APP] hook 加载失败：依赖缺失')
    return
  }

  /**
   * ⚠️ 以下数值均为测试默认值，正式节奏待定
   */
  const RHYTHM = {
    everyMessages: 3,        // 每 3 条 AI 消息推进一次
    minutesPerTick: 60,      // 每次推进 60 分钟
    useAITime: true          // AI 报了 <time> 就优先用
  }

  let _handler = null
  let _msgSeq = 0
  let _lastReport = null

  // ============================================================
  // 一、时间解析
  // ============================================================

  /**
   * 从 AI 消息里抠出 <time elapsed="30" />
   * @param {string} text
   * @returns {number|null}
   */
  function parseElapsed(text) {
    if (!text) return null
    const m = /<time[^>]*\belapsed\s*=\s*"(\d+)"/i.exec(text)
    if (!m) return null
    const n = Number(m[1])
    return Number.isFinite(n) && n >= 0 && n <= 1440 ? n : null
  }

  // ============================================================
  // 二、推进一步
  // ============================================================

  /**
   * 推进一步
   * @param {number} minutes 推进多少分钟（0 = 只应用申报，不动时间）
   * @param {object} opts { msgId, revision, force, text }
   * @returns {object} 报告
   */
  function step(minutes, opts = {}) {
    let world = W.ensureWorld()

    const text = opts.text || ''
    const P = root.OKH_APP_PARSE

    // 解析申报标签 → front
    const front = (P && text) ? P.buildFront(text, world.timeMin, world) : null
    const hasFront = front && Object.keys(front).length > 0

    const toMin = world.timeMin + Math.max(0, Math.floor(minutes))
    if (toMin === world.timeMin && !hasFront && !opts.force) {
      return { advanced: false, reason: '没有时间可推进，也没有申报' }
    }

    const id = opts.msgId || `tick-${Date.now()}`
    const revision = opts.revision || 'r1'

    // 幂等：同一条消息不重复推进
    if (world.bg.receipts && world.bg.receipts[id]) {
      return { advanced: false, reason: '重复提交' }
    }

    let result
    try {
      result = GPT.advance(world, {
        id,
        revision,
        toMin,
        interactingIds: front?.interactingIds || [],
        front: front || null
      }, CFG.buildConfig())
    } catch (e) {
      console.error('[APP] advance 失败：', e.message)
      return { advanced: false, reason: e.message }
    }

    if (result.duplicate) return { advanced: false, reason: '重复提交' }

    const state = result.state

    // 应用层：驱动力漂移（只在时间真的前进时做）
    if (toMin > world.timeMin) A.drift(state)

    // 存回
    W.saveWorld(state)

    // 待演绎文本
    const batch = GPT.prepare(state, 8)
    if (batch.ids.length) {
      W.store.set('okh_gpt.pendingText', batch.text)
      W.store.set('okh_gpt.pendingIds', JSON.stringify(batch.ids))
    }

    // 台账与监听
    W.store.set('okh_gpt.ledger', A.buildLedger(state))
    W.store.set('okh_gpt.watch', JSON.stringify(A.buildWatch(state)))

    _lastReport = {
      advanced: true,
      from: world.timeMin,
      to: state.timeMin,
      front: front || null,
      actions: result.report.actions,
      released: result.report.released,
      pending: batch.ids.length
    }

    if (typeof console !== 'undefined') {
      const frontInfo = hasFront ? ` · 申报 ${Object.keys(front).join(',')}` : ''
      console.log(`[APP] ${world.timeMin} → ${state.timeMin} · 行动 ${result.report.actions.length} 项 · 待演绎 ${batch.ids.length} 条${frontInfo}`)
    }

    return _lastReport
  }

  // ============================================================
  // 三、消息钩子
  // ============================================================

  /**
   * 收到 AI 消息
   *
   * 两套推进并存：
   *   A. AI 申报了 <time elapsed>  → 按申报推进（elapsed=0 时只应用申报，不动时间）
   *   B. AI 没申报               → 每 N 条消息推进一次固定时长
   */
  function onMessage() {
    try {
      const ctx = typeof SillyTavern !== 'undefined' ? SillyTavern.getContext() : null
      const chat = ctx?.chat
      if (!chat || !chat.length) return

      const last = chat[chat.length - 1]
      if (!last || last.is_user) return

      _msgSeq++
      const text = last.mes || ''
      const msgId = `msg-${chat.length}`
      const revision = text.slice(0, 32) || 'r1'

      // 解析 status_bar
      const P = root.OKH_APP_PARSE
      const parsed = P ? P.parseStatusBar(text) : { elapsed: parseElapsed(text) }
      const elapsed = parsed.elapsed

      // A. AI 报了时间（含 elapsed=0 的开场）
      if (elapsed !== null && RHYTHM.useAITime) {
        _msgSeq = 0
        step(elapsed, { msgId, revision, text })
        return
      }

      // B. 固定节奏
      if (_msgSeq >= RHYTHM.everyMessages) {
        _msgSeq = 0
        step(RHYTHM.minutesPerTick, { msgId, revision, text })
      }
    } catch (e) {
      console.error('[APP] 消息处理失败：', e)
    }
  }

  /**
   * 挂钩子
   * @returns {boolean}
   */
  function attach() {
    if (_handler) { console.log('[APP] 钩子已存在'); return true }
    if (typeof eventOn !== 'function' || typeof tavern_events === 'undefined') {
      console.warn('[APP] 当前环境不支持 eventOn，请手动调 OKH_APP.step()')
      return false
    }
    _handler = onMessage
    eventOn(tavern_events.MESSAGE_RECEIVED, _handler)
    console.log('[APP] 钩子已挂上')
    return true
  }

  /**
   * 摘钩子
   * @returns {boolean}
   */
  function detach() {
    if (!_handler) return false
    if (typeof eventRemoveListener === 'function' && typeof tavern_events !== 'undefined') {
      eventRemoveListener(tavern_events.MESSAGE_RECEIVED, _handler)
    }
    _handler = null
    console.log('[APP] 钩子已摘下')
    return true
  }

  root.OKH_APP_HOOK = Object.freeze({
    RHYTHM,
    parseElapsed,
    step,
    attach,
    detach,
    onMessage,
    lastReport: () => _lastReport
  })

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = root.OKH_APP_HOOK
  }
})(typeof globalThis !== 'undefined' ? globalThis : this)
;

/**
 * @module okh_app_hud
 * @updated 2026-11-09 07:45
 * @status  demo
 * @reason  把 world 渲染成界面：在场面板（玩家可见）+ 监听面板（隐藏）
 * @deps    ['okh_app_world', 'okh_app_adapter']
 * @provides { render, startHud, stopHud, LOCATION_NAMES }
 *
 * 用法：
 *   页面上放 <div data-okh-hud></div>  → 显示时间、位置、在场
 *   页面上放 <div data-okh-watch></div> → 显示全员监听（建议设 display:none）
 */

(function (root) {
  'use strict'

  const W = root.OKH_APP_WORLD
  const A = root.OKH_APP_ADAPTER

  if (!W || !A) {
    console.error('[APP] hud 加载失败：需要 02_world 与 03_adapter')
    return
  }

  /** 地点中文名 */
  const LOCATION_NAMES = {
    town: '镇上',
    inn: '断剑与麦酒',
    church: '圣辉大教堂',
    manor: '领主山庄',
    casino: '地下赌场',
    wilds: '城外荒原'
  }

  /** 时段中文名 */
  const PHASE_NAMES = [
    { max: 360, name: '深夜' },
    { max: 600, name: '上午' },
    { max: 1020, name: '午后' },
    { max: 1440, name: '夜晚' }
  ]

  /**
   * 由分钟推时段名
   * @param {number} local
   * @returns {string}
   */
  function phaseName(local) {
    return (PHASE_NAMES.find(p => local < p.max) || PHASE_NAMES[3]).name
  }

  /**
   * 地点名
   * @param {string} key
   * @returns {string}
   */
  function locName(key) {
    return LOCATION_NAMES[key] || key
  }

  // ============================================================
  // 一、在场面板
  // ============================================================

  /**
   * 生成在场面板的 HTML
   * @param {object} world
   * @returns {string}
   */
  function buildHudHtml(world) {
    if (!world) return ''

    const day = Math.floor(world.timeMin / 1440) + 1
    const local = world.timeMin % 1440
    const hour = String(Math.floor(local / 60)).padStart(2, '0')
    const minute = String(local % 60).padStart(2, '0')

    const here = Object.values(world.npcs)
      .filter(n => n.alive !== false && n.location === world.playerLocation)

    const presentText = here.length
      ? here.map(n => {
          const busy = n.activity ? `（${n.activity.label}）` : ''
          return `<span class="okh-name">${escape(n.name)}</span><span class="okh-busy">${escape(busy)}</span>`
        }).join('　')
      : '<span class="okh-none">无人</span>'

    return `
      <div class="okh-row">
        <span class="okh-label">时间</span>
        <span class="okh-value">第 ${day} 天 · ${phaseName(local)}</span>
        <span class="okh-clock">${hour}:${minute}</span>
      </div>
      <div class="okh-row">
        <span class="okh-label">位置</span>
        <span class="okh-value">${escape(locName(world.playerLocation))}</span>
      </div>
      <div class="okh-row">
        <span class="okh-label">在场</span>
        <span class="okh-value">${presentText}</span>
      </div>
    `.trim()
  }

  // ============================================================
  // 二、监听面板
  // ============================================================

  /**
   * 生成监听面板的 HTML（等宽文本，便于对齐）
   * @param {object} world
   * @returns {string}
   */
  function buildWatchHtml(world) {
    if (!world) return ''
    const STAGE = { 0: '正常', 1: '异常', 2: '沉沦', 3: '臣服' }

    const lines = Object.values(world.npcs).map(n => {
      const mark = n.location === world.playerLocation ? '●' : '○'
      const stage = STAGE[n.stage] || n.stage
      const busy = n.activity ? `｜${n.activity.label}` : ''
      return `${mark} ${n.name}｜${locName(n.location)}｜${stage}｜堕落 ${Math.round(n.values.corruption ?? 0)}${busy}`
    })

    return lines.join('\n')
  }

  /**
   * HTML 转义
   * @param {string} s
   * @returns {string}
   */
  function escape(s) {
    return String(s ?? '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
  }

  // ============================================================
  // 三、渲染
  // ============================================================

  /**
   * 渲染一次
   * @returns {boolean} 是否找到容器
   */
  function render() {
    if (typeof document === 'undefined') return false

    const world = W.loadWorld()
    if (!world) return false

    let found = false

    // 在场面板：可能有多个（每条消息一个），全部填
    const huds = document.querySelectorAll('[data-okh-hud]')
    for (const el of huds) {
      el.innerHTML = buildHudHtml(world)
      el.setAttribute('data-okh-ready', '1')
      found = true
    }

    // 监听面板
    const watches = document.querySelectorAll('[data-okh-watch]')
    for (const el of watches) {
      el.textContent = buildWatchHtml(world)
      found = true
    }

    return found
  }

  // ============================================================
  // 四、定时刷新
  // ============================================================

  let _timer = null

  /**
   * 启动定时刷新
   * @param {number} intervalMs
   */
  function startHud(intervalMs = 1000) {
    if (_timer) return
    _timer = setInterval(() => {
      try { render() } catch (e) { /* 静默，不打断页面 */ }
    }, intervalMs)
    console.log('[APP] HUD 已启动 · 每 ' + intervalMs + 'ms 刷新')
  }

  /**
   * 停止
   */
  function stopHud() {
    if (!_timer) return
    clearInterval(_timer)
    _timer = null
    console.log('[APP] HUD 已停止')
  }

  root.OKH_APP_HUD = Object.freeze({
    LOCATION_NAMES,
    phaseName,
    locName,
    buildHudHtml,
    buildWatchHtml,
    render,
    startHud,
    stopHud
  })

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = root.OKH_APP_HUD
  }
})(typeof globalThis !== 'undefined' ? globalThis : this)
;

 * @module okh_app_entry

 * @status  demo

 * @deps    ['okh_app_config', 'okh_app_world', 'okh_app_adapter', 'okh_app_hook', 'okh_app_debug']
 * @provides { OKH_APP }

(function (root) {
  'use strict'

  const CFG = root.OKH_APP_CONFIG
  const W = root.OKH_APP_WORLD
  const A = root.OKH_APP_ADAPTER
  const H = root.OKH_APP_HOOK
  const D = root.OKH_APP_DEBUG
  const HUD = root.OKH_APP_HUD

  if (!CFG || !W || !A || !H || !D) {
    console.error('[APP] entry 加载失败：依赖缺失，检查 01~05 是否都加载了')

  }
  if (!HUD) {
    console.warn('[APP] entry：HUD 模块未加载，界面将不可用')
  }

  /**
   * 初始化世界
   * @param {boolean} force 为 true 时清空重建
   * @returns {object}
   */
  function init(force = false) {
    return W.ensureWorld(force)
  }

  /**
   * 启动：初始化 + 挂钩子 + 开 HUD
   * @returns {object} { world, hooked, hud }
   */
  function start() {
    const world = init(false)
    const hooked = H.attach()
    if (HUD) HUD.startHud(1000)
    console.log('[APP] 已启动 · ' + (hooked ? '自动推进已开启' : '需手动调 step()'))
    return { world, hooked, hud: !!HUD }

  /**
   * 停止：摘钩子 + 停 HUD，不清数据

  function stop() {
    H.detach()
    if (HUD) HUD.stopHud()
    console.log('[APP] 已停止')
  }
