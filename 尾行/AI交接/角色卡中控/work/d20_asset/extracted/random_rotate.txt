def write_json(obj, file):
    with open(file, 'w') as outfile:
        json.dump(obj, outfile)
    
    
def load_json(file):
    script_dir = os.path.dirname(__file__) #<-- absolute dir the script is in
    
    #open the json file
    if os.path.isabs(file):
        pass
    else:
        file = os.path.join(script_dir, file)
    
    data_file = open(file)
    obj = json.load(data_file)
    return dict(obj)

def update_dict(roll='last_roll', print_all=False):
    print('Updating dictionary...')
    bpy.ops.screen.animation_cancel(restore_frame=True)
    file = os.path.join(bpy.path.abspath("//"), 'data.json')
    roll = str(roll).zfill(2)

    obj = load_json(file)
    obj[roll] = {u"rotation": bpy.context.active_object.rotation_euler[0:3], u"location": bpy.context.active_object.location[0:3]}

    if print_all:
        print(json.dumps(obj, indent=4))
    else:
        print('Roll:', roll)
        print(json.dumps(obj[roll], indent=4))


    write_json(obj, file)
    print('Done!')
    
def roll_in_dict(new_roll):
    print('------')
    new_roll = str(new_roll).zfill(2)
    print('Checking dictionary for the roll:', new_roll)
    
    file = os.path.join(bpy.path.abspath("//"), 'data.json')
    obj = load_json(file)
    if new_roll in obj.keys():
        print('Key already exist. Current keys:', sorted(obj.keys()))
        rnd_rotate_stopped()
        
    else:
        print('Key not found.')
        update_dict(new_roll)
    print('------')
    
    
def rollmea(new_roll):

    print('------')
    print('Doing a new roll of a', new_roll)

    file = os.path.join(bpy.path.abspath("//"), 'data.json')
    obj = load_json(file)
    new_roll = str(new_roll).zfill(2)
    print('Parameters:', obj[new_roll])
    print('Please wait for the renderization...')


    bpy.context.active_object.delta_rotation_euler = (0.0, 0.0, 0.0)
    obj[new_roll]['rotation'][0]
    rotation = [

        obj[new_roll]['rotation'][1],
        obj[new_roll]['rotation'][2]    
    ]
    location = [
        obj[new_roll]['location'][0],
        obj[new_roll]['location'][1],
        obj[new_roll]['location'][2]    
    ]
    

    bpy.context.active_object.rotation_euler = (rotation[0], rotation[1], rotation[2])

    bpy.context.scene.frame_set(1) 
    bpy.ops.screen.animation_cancel(restore_frame=True)
    
    frame_num = 1
    last_frame = bpy.context.scene.frame_end
    for frame in range(1, (last_frame +1)):
        bpy.context.scene.frame_set(frame_num)    
        frame_num += 1
    #bpy.ops.render.render()
    print('Done!')
    print('------')

def rnd_rotate():


    bpy.context.active_object.delta_rotation_euler = (0.0, 0.0, 0.0)

    rotation = [
        random.randrange(0, 359 +1),
        random.randrange(0, 359 +1),
        random.randrange(0, 359 +1)    
    ]
    location = random.choice(START_POS)
    

    bpy.context.active_object.rotation_euler = (rotation[0], rotation[1], rotation[2])
    bpy.context.active_object.location = (location[0], location[1], location[2])
    
    bpy.context.scene.frame_set(1) 
    bpy.ops.screen.animation_cancel(restore_frame=True)
    
    bpy.ops.screen.animation_play()
